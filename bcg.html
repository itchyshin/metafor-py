Does BCG vaccination reduce tuberculosis risk?

The BCG example asks whether vaccination lowers tuberculosis incidence across 13 studies, and whether the effect changes with absolute latitude. The published example provides the study counts and describes the motivating question. This page uses the shared data fixture and original prose. Dataset and upstream example

Data

Each study reports tuberculosis cases and non-cases in vaccinated and control groups. We calculate the log risk ratio from those counts. Negative values indicate a lower risk among vaccinated participants.

"""Run a focused stage or the complete 13-study BCG example."""
import argparse
import csv
import json
from pathlib import Path

import numpy as np

from metafor import escalc_rr, fit_meta, forest_plot_data, write_forest_svg

ROOT = Path(__file__).parents[1]


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--case",
        choices=("all", "effects", "models", "moderator", "prediction", "forest", "table-adapter"),
        default="all",
    )
    case = parser.parse_args().case

    with (ROOT / "reference/v1/bcg.csv").open(newline="") as handle:
        rows = list(csv.DictReader(handle))
    count = lambda name: np.array([float(row[name]) for row in rows])
    yi, vi = escalc_rr(count("tpos"), count("tneg"), count("cpos"), count("cneg"))
    latitude = count("ablat")
    if case == "table-adapter":
        import pandas as pd
        from metafor import fit_meta_formula

        trials = np.array([int(row["trial"]) for row in rows])
        table = pd.DataFrame({"yi": yi, "vi": vi, "ablat": latitude}, index=trials)
        table.loc[trials[3], "ablat"] = np.nan
        formula_fit = fit_meta_formula(table, "yi ~ 1 + ablat")
        keep = np.isfinite(table["ablat"].to_numpy())
        design = np.column_stack([np.ones(keep.sum()), table.loc[keep, "ablat"].to_numpy()])
        matrix_fit = fit_meta(yi[keep], vi[keep], design)
        max_diff = float(np.max(np.abs(formula_fit.result.beta - matrix_fit.beta)))
        if formula_fit.row_index != tuple(trials[keep]) or max_diff > 1e-12:
            raise RuntimeError("table adapter changed row identity or disagreed with matrix input")
        fmt = lambda values: ",".join(f"{float(value):.16g}" for value in np.atleast_1d(values))
        print(f"TABLE_ADAPTER|design_columns|{','.join(formula_fit.design_columns)}")
        print(f"TABLE_ADAPTER|row_ids|{','.join(str(x) for x in formula_fit.row_index)}")
        print(f"TABLE_ADAPTER|design_shape|{design.shape[0]},{design.shape[1]}")
        print(f"TABLE_ADAPTER|design_values|{fmt(design.ravel())}")
        print(f"TABLE_ADAPTER|beta|{fmt(formula_fit.result.beta)}")
        print(f"TABLE_ADAPTER|matrix_max_abs_diff|{max_diff:.16g}")
        print("TABLE_ADAPTER_OK")
        return
    designs = {
        "fixed": np.ones((len(yi), 1)),
        "random": np.ones((len(yi), 1)),
        "latitude": np.column_stack([np.ones(len(yi)), latitude]),
    }
    requested = {
        "all": ("fixed", "random", "latitude"),
        "effects": (),
        "models": ("fixed", "random"),
        "moderator": ("latitude",),
        "prediction": ("latitude",),
        "forest": ("random",),
    }[case]
    models = {
        name: fit_meta(yi, vi, designs[name] if name == "latitude" else None,
                       method="FE" if name == "fixed" else "REML" if name == "random" else "REML")
        for name in requested
    }
    predictions = None
    if case in ("all", "moderator", "prediction"):
        predictions = models["latitude"].predict(
            np.column_stack([np.ones(3), [0.0, 30.0, 60.0]])
        )
    forest = None
    if case in ("all", "forest"):
        forest = forest_plot_data(yi, vi, [row["author"] for row in rows], models["random"])

    output = {
        name: {
            "beta": model.beta.tolist(),
            "se": model.se.tolist(),
            "tau2": model.tau2,
            "QE": model.q,
            "I2": model.i2,
            "H2": model.h2,
            "ci_lb": model.interval()[0].tolist(),
            "ci_ub": model.interval()[1].tolist(),
        }
        for name, model in models.items()
    }
    if predictions is not None:
        output["prediction"] = {
            "latitude": [0.0, 30.0, 60.0],
            "estimate": predictions.estimate.tolist(),
            "se": predictions.se.tolist(),
            "ci_lb": predictions.ci_lb.tolist(),
            "ci_ub": predictions.ci_ub.tolist(),
            "pi_lb": predictions.pi_lb.tolist(),
            "pi_ub": predictions.pi_ub.tolist(),
        }
    output["yi"] = yi.tolist()
    output["vi"] = vi.tolist()
    result_path = (
        ROOT / "examples/bcg-results.json"
        if case == "all"
        else ROOT / "examples/generated/bcg" / f"{case}-results.json"
    )
    result_path.parent.mkdir(parents=True, exist_ok=True)
    result_path.write_text(json.dumps(output, indent=2) + "\n")

    def fmt(values):
        return ",".join(f"{float(value):.16g}" for value in np.atleast_1d(values))

    if case in ("all", "effects"):
        for i, (effect, variance) in enumerate(zip(yi, vi), start=1):
            print(f"EFFECT|{i}|{effect:.16g}|{variance:.16g}")
    coefficient_labels = {"fixed": ["intrcpt"], "random": ["intrcpt"],
                          "latitude": ["intrcpt", "ablat"]}
    for name, model in models.items():
        ci_lb, ci_ub = model.interval()
        print("RESULT|{}|beta={}|se={}|tau2={:.16g}|QE={:.16g}|I2={:.16g}|H2={:.16g}|ci_lb={}|ci_ub={}".format(
            name, fmt(model.beta), fmt(model.se), model.tau2, model.q,
            model.i2, model.h2, fmt(ci_lb), fmt(ci_ub)))
        print(f"COV|{name}|{fmt(model.covariance.ravel())}")
        print(f"COVLABELS|{name}|{','.join(coefficient_labels[name])}")
        print(f"ROWS|{name}|{','.join(str(i + 1) for i in model.row_indices)}")
        print(f"DESIGN|{name}|{len(designs[name])}|{designs[name].shape[1]}|{fmt(designs[name].ravel())}")
    if predictions is not None:
        for i, value in enumerate((0.0, 30.0, 60.0)):
            print("PRED|latitude|x={:.16g}|estimate={:.16g}|se={:.16g}|ci_lb={:.16g}|ci_ub={:.16g}|pi_lb={:.16g}|pi_ub={:.16g}".format(
                value, predictions.estimate[i], predictions.se[i], predictions.ci_lb[i],
                predictions.ci_ub[i], predictions.pi_lb[i], predictions.pi_ub[i]))
    if forest is not None:
        forest_json = ROOT / "examples/bcg-forest-data.json"
        forest_json.write_text(json.dumps(forest, indent=2) + "\n")
        svg_path = write_forest_svg(forest, ROOT / "examples/bcg-forest.svg")
        if len(forest["rows"]) != 13 or svg_path.stat().st_size == 0:
            raise RuntimeError("forest plot output is incomplete")
        summary = forest["summary"]
        print("FOREST_SUMMARY|estimate={estimate:.16g}|se={se:.16g}|ci_lb={ci_lb:.16g}|ci_ub={ci_ub:.16g}|pi_lb={pi_lb:.16g}|pi_ub={pi_ub:.16g}|tau2={tau2:.16g}".format(**summary))
        print(f"FOREST_PLOT_OK {svg_path}")
    print(f"BCG_EXAMPLE_OK case={case} {result_path}")


if __name__ == "__main__":
    main()

Models

The script fits an inverse-variance fixed-effect model, a REML random-effects model, and a REML model with absolute latitude as a moderator. REML allows underlying study effects to vary around a mean. The moderator coefficient is the estimated change in log risk ratio per degree of latitude.

Interpretation

The pooled log risk ratio is negative, consistent with lower tuberculosis risk among vaccinated participants in these studies. The negative latitude coefficient indicates a more negative estimated log risk ratio at higher absolute latitudes. These are associations in this set of studies, and the moderator result alone does not establish why effects vary geographically.

The example also predicts mean log risk ratios at 0°, 30°, and 60° latitude. The confidence interval describes uncertainty in the fitted mean. The prediction interval is wider because it includes both uncertainty in that mean and the estimated variation among study effects. It writes the numeric results, the row-aligned forest data, and an original SVG plot under examples/.

Supplied phylogenetic correlation

The next runnable example uses five synthetic observations on three species and a supplied labelled species-correlation matrix. It includes an exact zero-boundary case and an interior variance-component case, each checked against the pinned R reference. The fixtures are for software parity, not empirical biological claims.

"""Fit boundary and interior synthetic phylogenetic examples."""
import argparse
import csv
import json
from pathlib import Path
import numpy as np
from metafor import fit_phylo

ROOT = Path(__file__).parents[1]
parser = argparse.ArgumentParser(description="Run a supplied-correlation phylogenetic fixture.")
parser.add_argument("--case", choices=("all", "boundary", "interior"), default="all")
case_arg = parser.parse_args().case
case_for_fixture = {"boundary": "v1", "interior": "v2-interior"}
results = {}
fixtures = ("v1", "v2-interior") if case_arg == "all" else (case_for_fixture[case_arg],)
for case in fixtures:
    base = ROOT / "reference/phylogeny" / case
    observations = list(csv.DictReader((base / "observations.csv").open()))
    matrix_rows = list(csv.reader((base / "species-correlation.csv").open(newline="")))
    labels = matrix_rows[0][1:]
    if [row[0] for row in matrix_rows[1:]] != labels:
        raise ValueError("correlation matrix row and column labels must match")
    A = np.asarray([[float(x) for x in row[1:]] for row in matrix_rows[1:]])
    fit = fit_phylo([float(r["yi"]) for r in observations],
                    [float(r["vi"]) for r in observations],
                    [r["species"] for r in observations], A, labels)
    low, high = fit.interval()
    results[case] = {"beta": float(fit.beta[0]), "se": float(fit.se[0]),
                     "tau2": float(fit.tau2), "ci_lb": float(low[0]),
                     "ci_ub": float(high[0])}
    values = results[case]
    print("PHYLO|{}|beta={:.16g}|se={:.16g}|tau2={:.16g}|ci_lb={:.16g}|ci_ub={:.16g}".format(
        case, values["beta"], values["se"], values["tau2"], values["ci_lb"], values["ci_ub"]))
out = (
    ROOT / "examples/phylogeny-results.json"
    if case_arg == "all"
    else ROOT / "examples/generated/phylogeny" / f"{case_arg}-results.json"
)
out.parent.mkdir(parents=True, exist_ok=True)
out.write_text(json.dumps(results, indent=2) + "\n", encoding="utf-8")
print(f"PHYLOGENY_EXAMPLE_OK {out}")

The paired report checks the supplied-correlation fits against the pinned R reference. Tree estimation and prediction for unseen species are not part of this example. The correlation matrix and observation order are supplied explicitly; see the R and Python comparison guide for checked quantities and tolerances.

Differences from R metafor

This experimental Python port supports log risk ratios, matrix and optional pandas/Patsy formula inputs, fixed-effect and REML models, normal mean and new-study prediction intervals, forest data, and the supplied-correlation phylogenetic model shown here. It omits non-finite rows with a runtime warning and preserves row indices. Warning wording and categories differ from R. REML with zero sampling variances is unsupported; the fit raises an explicit error instead of silently discarding those rows. Full R formula semantics, other effect-size measures, multivariate structures, and many diagnostic and inference options remain unsupported. See the compatibility table.
